-- 01_express_checkout / express_checkout_shown
CREATE TABLE atlys.express_checkout_shown
(
    `os` String,
    `user_id` String,
    `currency` String,
    `timestamp` DateTime64(3),
    `app_version` String,
    `destination` String,
    `device_type` String,
    `shown_amount` String,
    `application_id` String,
    `geoip_country_code` String
)
ENGINE = SharedMergeTree('/clickhouse/tables/{uuid}/{shard}', '{replica}')
PARTITION BY toYYYYMM(timestamp)
ORDER BY (toDate(timestamp), device_type, destination, user_id, application_id)
SETTINGS index_granularity = 8192
-- MV for express_checkout_shown
CREATE MATERIALIZED VIEW atlys.mv_express_checkout_shown_to_activity TO atlys.activity_events
(
    `event_name` String,
    `ch_table` String,
    `timestamp` DateTime64(3),
    `user_id` String,
    `application_id` String,
    `device_type` String,
    `os` String,
    `geoip_country_code` String,
    `destination` String,
    `payload` String
)
AS SELECT
    'express_checkout_shown' AS event_name,
    'express_checkout_shown' AS ch_table,
    timestamp AS timestamp,
    toString(user_id) AS user_id,
    toString(application_id) AS application_id,
    toString(device_type) AS device_type,
    toString(os) AS os,
    toString(geoip_country_code) AS geoip_country_code,
    toString(destination) AS destination,
    toJSONString(mapFromArrays(['os', 'user_id', 'currency', 'timestamp', 'app_version', 'destination', 'device_type', 'shown_amount', 'application_id', 'geoip_country_code'], [toString(os), toString(user_id), toString(currency), toString(timestamp), toString(app_version), toString(destination), toString(device_type), toString(shown_amount), toString(application_id), toString(geoip_country_code)])) AS payload
FROM atlys.express_checkout_shown
